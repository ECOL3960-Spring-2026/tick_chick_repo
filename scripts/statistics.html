Test assumptions of linear model and diagnostic plots

# residuals vs. fitted plot
plot(ln_model, which = 1)

# QQ residuals plot
plot(ln_model, which = 2)

# Scale-location plot
plot(ln_model, which = 3)

# Shapiro test of linear model residuals
shapiro.test(residuals(ln_model))
## 
##  Shapiro-Wilk normality test
##

Plot Linear Regression

lm_plot <- ggplot(pathogenAndTemp,
                  aes(x = mean_temp,
                      y = pathogenCount)) +
  geom_point(alpha = 0.6) + 
  geom_smooth(method = "lm", se = TRUE, color = "seagreen") +
  theme_classic() +
  labs(
    x = "Mean Daily Temperature (ºC)",
    y = "Pathogen Count",
    title = "Relationship of Daily Temperature and Pathogen Count at HARV Site"
  )
